cmake_minimum_required(VERSION 3.21)
project(TurboInteractInstalledConsumers LANGUAGES C CXX)
find_package(TurboInteract CONFIG REQUIRED)
add_executable(ti_c_consumer c_consumer.c)
add_executable(ti_cpp_consumer cpp_consumer.cpp)
target_compile_features(ti_c_consumer PRIVATE c_std_11)
target_compile_features(ti_cpp_consumer PRIVATE cxx_std_17)
target_link_libraries(ti_c_consumer PRIVATE TurboInteract::Core)
target_link_libraries(ti_cpp_consumer PRIVATE TurboInteract::Core)
enable_testing()
add_test(NAME ti_c_consumer COMMAND ti_c_consumer)
add_test(NAME ti_cpp_consumer COMMAND ti_cpp_consumer)
